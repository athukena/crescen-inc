// ============================================================ // Edge Function: shipping-quote — Crescen Inc // Cotiza el envío con Skydropx SIN exponer las llaves al navegador. // ============================================================ const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", }; // Origen fijo: la bodega/sucursal principal de Crescen. Ajusta si hace falta. const ORIGIN = { postal_code: "06000", // CP de Plaza de la Tecnología area_level1: "Ciudad de Mexico", area_level2: "Cuauhtemoc", area_level3: "Centro (Area 1)", country_code: "MX", }; // Paquete estándar para un pedido de cartuchos — ajustable según lo que pese en promedio. const DEFAULT_PARCEL = { weight: 2, // kg length: 30, // cm width: 25, height: 15, }; const SKYDROPX_BASE = "https://api-pro.skydropx.com"; // Margen de seguridad sobre el precio que cotiza Skydropx, para cubrir // variaciones de peso/tamaño reales al momento de embalar. Ajusta este // número cuando quieras — 1.30 = 30% arriba del precio cotizado. const SHIPPING_MARKUP = 1.30; async function getAccessToken(): Promise { const clientId = Deno.env.get("SKYDROPX_CLIENT_ID"); const clientSecret = Deno.env.get("SKYDROPX_CLIENT_SECRET"); if (!clientId || !clientSecret) { throw new Error("Faltan las credenciales de Skydropx en los secrets de Supabase"); } const res = await fetch(`${SKYDROPX_BASE}/api/v1/oauth/token`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ client_id: clientId, client_secret: clientSecret, grant_type: "client_credentials", }), }); if (!res.ok) { const bodyText = await res.text(); throw new Error(`No se pudo autenticar con Skydropx (status ${res.status}): ${bodyText}`); } const data = await res.json(); return data.access_token; } async function createQuotation(token: string, dest: { postal_code: string; colonia?: string; municipio?: string; estado?: string }) { const addressTo: Record = { postal_code: dest.postal_code, country_code: "MX", }; if (dest.estado) addressTo.area_level1 = dest.estado; if (dest.municipio) addressTo.area_level2 = dest.municipio; if (dest.colonia) addressTo.area_level3 = dest.colonia; const res = await fetch(`${SKYDROPX_BASE}/api/v1/quotations`, { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${token}`, }, body: JSON.stringify({ quotation: { address_from: ORIGIN, address_to: addressTo, parcel: DEFAULT_PARCEL, }, }), }); if (!res.ok) { const bodyText = await res.text(); throw new Error(`No se pudo crear la cotización con Skydropx (status ${res.status}): ${bodyText}`); } const data = await res.json(); return data.id; } async function pollQuotation(token: string, quotationId: string) { // Las cotizaciones se completan progresivamente — los primeros "rates" pueden // venir con price:0 mientras se calculan. Esperamos a que is_completed sea true // Y a que haya al menos una tarifa con precio real mayor a 0. let lastRates: any[] = []; for (let i = 0; i < 10; i++) { const res = await fetch(`${SKYDROPX_BASE}/api/v1/quotations/${quotationId}`, { headers: { "Authorization": `Bearer ${token}` }, }); const data = await res.json(); const rates = data?.rates || data?.data?.rates || []; lastRates = rates; const hasRealPrice = rates.some((r: any) => Number(r.total ?? r.price) > 0); if (data.is_completed && hasRealPrice) { return rates; } await new Promise((r) => setTimeout(r, 1500)); } // Si se acabaron los intentos, regresa lo que haya (aunque sea parcial) en vez de nada return lastRates.filter((r: any) => Number(r.total ?? r.price) > 0); } Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } try { const { postal_code, colonia, municipio, estado } = await req.json(); if (!postal_code || !/^\d{5}$/.test(postal_code)) { return new Response(JSON.stringify({ error: "Código postal inválido" }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } const token = await getAccessToken(); const quotationId = await createQuotation(token, { postal_code, colonia, municipio, estado }); const rates = await pollQuotation(token, quotationId); if (rates.length === 0) { return new Response(JSON.stringify({ error: "No hay opciones de envío para ese código postal" }), { status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } // Ordena por precio y regresa la más barata + un par de alternativas const validRates = rates.filter((r: any) => r.total != null || r.price != null); if (validRates.length === 0) { return new Response(JSON.stringify({ error: "Las paqueterías aún no devolvieron precios completos, intenta de nuevo" }), { status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } const sorted = validRates.sort((a: any, b: any) => Number(a.total ?? a.price) - Number(b.total ?? b.price)); const options = sorted.slice(0, 3).map((r: any) => ({ carrier: r.carrier_name || r.provider_name, service: r.service_level_name || r.service, price: Math.round(Number(r.total ?? r.price) * SHIPPING_MARKUP), days: r.days || null, })); return new Response(JSON.stringify({ options, cheapest: options[0] }), { headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } catch (err) { return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } });
